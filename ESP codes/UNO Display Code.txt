// ── RGB565 palette ──────────────────────────────────────────────────────────

#define CLR_PINK      0xF1EB   // Lyra rose
#define CLR_PURPLE    0x799D   // Lyra purple

// Buffer size = ((4*version+17)^2 + 7) / 8 + 1 = ((29*29)+7)/8+1 = 107 for version 3
#define QR_BUF_SIZE 107
static QRCode  g_qr;
static uint8_t g_qrBuf[QR_BUF_SIZE];

// ── Screen state — skip full redraw if same screen requested again ──────────
static char g_curScreen = 0;

// ── Primitive helpers ───────────────────────────────────────────────────────
void bg(uint16_t col = CLR_BG)     { tft.fillScreen(col); }
void hline(int y, uint16_t c)      { tft.drawFastHLine(0, y, W, c); }
void topBar(int h, uint16_t c)     { tft.fillRect(0, 0, W, h, c); }
void bottomBar(int h, uint16_t c)  { tft.fillRect(0, H - h, W, h, c); }
void pill(int x,int y,int w,int h,uint16_t c){ tft.fillRoundRect(x,y,w,h,h/2,c); }

// Card with subtle shadow
void card(int x, int y, int w, int h, uint16_t bg_col, uint16_t border_col) {
  tft.fillRoundRect(x + 2, y + 3, w, h, 8, CLR_BLACK);  // shadow
  tft.fillRoundRect(x, y, w, h, 8, bg_col);
  tft.drawRoundRect(x, y, w, h, 8, border_col);
}

// Modern progress bar: track + fill + inner shine
void progressBar(int x, int y, int w, int h, int pct, uint16_t col) {
  tft.fillRoundRect(x, y, w, h, h/2, CLR_DIMGRAY);
  int fw = max(h, (w * pct) / 100);
  tft.fillRoundRect(x, y, fw, h, h/2, col);
  if (fw > 6) tft.drawFastHLine(x + 3, y + 2, fw - 6, (uint16_t)(col | 0x1082));
}

// Glow ring — 2 rings only (was 4; each drawCircle is O(r) pixel writes)
void glowCircle(int cx, int cy, int r, uint16_t dim_col) {
  tft.drawCircle(cx, cy, r + 1, dim_col);
  tft.drawCircle(cx, cy, r + 2, dim_col);
}

// Centre-print Flash string
void centreText(const __FlashStringHelper* s, int y, uint16_t col, uint8_t sz,
                uint16_t bg_col = CLR_BG) {
  tft.setTextColor(col, bg_col);
  tft.setTextSize(sz);
  int len = 0;
  PGM_P p = reinterpret_cast<PGM_P>(s);
  while (pgm_read_byte(p++)) len++;
  tft.setCursor(max(0, (W - len * 6 * sz) / 2), y);
  tft.print(s);
}

// Centre-print RAM string
void centreText(const char* s, int y, uint16_t col, uint8_t sz,
                uint16_t bg_col = CLR_BG) {
  tft.setTextColor(col, bg_col);
  tft.setTextSize(sz);
  tft.setCursor(max(0, (W - (int)strlen(s) * 6 * sz) / 2), y);
  tft.print(s);
}

// Thick check mark — fixed 2-px thickness (5 lines), not radius-scaled loop
void drawCheck(int cx, int cy, int r, uint16_t col) {
  for (int i = -2; i <= 2; i++) {
    tft.drawLine(cx - r, cy + i, cx - r/4, cy + r*3/4 + i, col);
    tft.drawLine(cx - r/4, cy + r*3/4 + i, cx + r, cy - r/2 + i, col);
  }
}

// Thick X — fixed 2-px thickness
void drawX(int cx, int cy, int r, uint16_t col) {
  for (int i = -2; i <= 2; i++) {
    tft.drawLine(cx - r + i, cy - r, cx + r + i, cy + r, col);
    tft.drawLine(cx + r + i, cy - r, cx - r + i, cy + r, col);
  }
}

// Down arrow
void drawDownArrow(int cx, int cy, int sz, uint16_t col) {
  tft.fillTriangle(cx, cy + sz, cx - sz, cy - sz/2, cx + sz, cy - sz/2, col);
  tft.fillRect(cx - sz/3, cy - sz, sz*2/3, sz/2 + 2, col);
}

// WiFi arcs (3 arcs + dot, centred at cx,cy)
void drawWifi(int cx, int cy, uint16_t col) {
  tft.fillCircle(cx, cy + 2, 4, col);
  tft.drawCircle(cx, cy + 2, 12, col); tft.drawCircle(cx, cy + 2, 13, col);
  tft.drawCircle(cx, cy + 2, 22, col); tft.drawCircle(cx, cy + 2, 23, col);
  tft.drawCircle(cx, cy + 2, 32, col); tft.drawCircle(cx, cy + 2, 33, col);
  // Mask bottom half
  tft.fillRect(cx - 40, cy + 4, 80, 40, CLR_BG);
}

// ── QR Screen ─────────────────────────────────────────────────────────────
void showQRReady() {
  // g_qr is pre-computed in setup() — no Reed-Solomon math here
  uint8_t qrSize  = g_qr.size;
  uint8_t modSize = (W - 20) / (qrSize + 8);
  if (modSize < 2) modSize = 2;
  if (modSize > 7) modSize = 7;

  int qrPx    = qrSize  * modSize;
  int margin  = 4       * modSize;
  int totalPx = qrPx    + margin * 2;

  bg();
  tft.fillRect(0, 0, W/2, 5, CLR_PINK);
  tft.fillRect(W/2, 0, W - W/2, 5, CLR_PURPLE);
  centreText(F("LYRA"), 14, CLR_PINK, 3);
  centreText(F("Scan to Pay"), 47, CLR_GRAY, 1);

  // QR panel with border
  int panelX = (W - totalPx) / 2;
  int panelY = 64;
  tft.fillRoundRect(panelX - 4, panelY - 4, totalPx + 8, totalPx + 8, 6, CLR_GRAY);
  tft.fillRect(panelX, panelY, totalPx, totalPx, CLR_WHITE);

  int qx = panelX + margin;
  int qy = panelY + margin;
  // Batch consecutive dark modules per row → far fewer fillRect calls
  for (uint8_t row = 0; row < qrSize; row++) {
    int spanStart = -1;
    for (uint8_t col = 0; col <= qrSize; col++) {
      bool dark = (col < qrSize) && qrcode_getModule(&g_qr, col, row);
      if (dark && spanStart < 0) {
        spanStart = col;
      } else if (!dark && spanStart >= 0) {
        tft.fillRect(qx + spanStart * modSize, qy + row * modSize,
                     (col - spanStart) * modSize, modSize, CLR_BLACK);
        spanStart = -1;
      }
    }
  }

  int belowY = panelY + totalPx + 12;
  centreText(F("lyra-app.co.in"), belowY, CLR_GRAY, 1);

  tft.fillRect(0, H - 28, W, 28, CLR_PURPLE);
  centreText(F("Insert coin or scan above"), H - 19, CLR_WHITE, 1, CLR_PURPLE);
}

// ── Command dispatcher ─────────────────────────────────────────────────────
void processCommand(char c) {
  if (c == g_curScreen) return;  // already on this screen — skip full redraw
  g_curScreen = c;
  Serial.print(F("CMD: ")); Serial.println(c);
  switch (c) {
    case '0': showBoot();          break;
    case '1': showThankYou();      break;
    case '2': showQRReady();       break;
    case '3': showDispensing();    break;
    case '4': showCoinInserted();  break;
    case '5': showCollect();       break;
    case '6': showNetworkError();  break;
    case '8': showStockReset();    break;
    case '9': showOutOfStock();    break;
  }
}

// ── Setup & Loop ───────────────────────────────────────────────────────────
void setup() {
  Serial.begin(9600);
  espSerial.begin(9600);

  uint16_t ID = tft.readID();
  if (ID == 0xD3D3) ID = 0x9486;
  tft.begin(ID);
  tft.setRotation(0);
  W = tft.width();
  H = tft.height();

  // Generate QR once — Reed-Solomon math runs here, not on every screen switch
  qrcode_initText(&g_qr, g_qrBuf, 3, ECC_LOW, QR_URL);

  tft.fillScreen(CLR_BLACK);
  showBoot();
  delay(1500);
  g_curScreen = '2';
  showQRReady();
  Serial.println(F("UNO ready — waiting for ESP32 commands"));
}

void loop() {
  // Buffer-based line reader — handles \r\n from ESP32 cleanly
  static char espBuf[8];
  static uint8_t espLen = 0;

  while (espSerial.available()) {
    char c = espSerial.read();
    if (c == '\n' || c == '\r') {
      if (espLen > 0) {
        processCommand(espBuf[0]);
        espLen = 0;
      }
    } else if (espLen < (uint8_t)(sizeof(espBuf) - 1)) {
      espBuf[espLen++] = c;
    }
  }

  // USB Serial manual test — same logic
  static char usbBuf[8];
  static uint8_t usbLen = 0;

  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      if (usbLen > 0) {
        processCommand(usbBuf[0]);
        usbLen = 0;
      }
    } else if (usbLen < (uint8_t)(sizeof(usbBuf) - 1)) {
      usbBuf[usbLen++] = c;
    }
  }
}
